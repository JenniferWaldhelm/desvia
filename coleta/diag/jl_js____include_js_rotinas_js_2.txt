String.prototype.lpad = function (pSize, pCharPad) {
  var str = this;
  var dif = pSize - str.length;
  var ch = String(pCharPad).charAt(0);
  for (; dif > 0; dif--) str = ch + str;
  return (str);
};  //String.lpad

String.prototype.trim = function () {
  return this.replace(/^\s*/, "").replace(/\s*$/, "");
}; //String.trim

String.prototype.stripBlanks = function () {
  return this.replace(/^\s*/, "").replace(/\s/g, '');
}; //String.stripBlanks

Array.prototype.inArray = function (value) {
  var i;
  for (i = 0; i < this.length; i++) {
    if (this[i] === value) return true;
  }
  return false;
}

function dataValida(e) {
  var divisor = '/';
  var data = e.value;
  var dataAtual = '';

  if (data.match(/^(0[1-9]|[12][0-9]|3[01]).?(0[1-9]|1[012]).?([12][0-9]{3}|[0-9]{2})$/)) {
    data = data.replace(/[^0-9]/g, '');
    dataAtual = data.substr(0, 2) + divisor + data.substr(2, 2) + divisor;
    if (data.substr(4).length == 4) dataAtual += data.substr(4);
    else dataAtual += (data.substr(4) > 30 ? '19' : '20') + data.substr(4);
  }
  if (dataAtual != '') {
    x = dataAtual.split(divisor);
    confere = new Date(x[2], x[1] - 1, x[0]);
    confere2 = (confere.getDate() < 10 ? '0' : '') + confere.getDate();
    confere2 += divisor + ((confere.getMonth() + 1) < 10 ? '0' : '') + (confere.getMonth() + 1);
    confere2 += divisor + confere.getFullYear();
    if (confere2 != dataAtual) dataAtual = '';
  }
  if ((typeof dataAtual == 'undefined') || (dataAtual == ""))
    return false;
  else
    return true;
}

function inverterData3(data, sep) {
  var posSep = data.search(/[^0-9]/);
  var resultado = "";
  if (posSep == 2 || posSep == 4) {
    var sepAtual = data.substr(posSep, 1);
    var partes = data.split(sepAtual);
    resultado = partes[2] + sep + partes[1] + sep + partes[0];
  } else
    resultado = data;

  return resultado;
}

function getXHR() {
  var newReq = null;
  if (window.XMLHttpRequest) {
    try {
      newReq = new XMLHttpRequest();
    } catch (e) {
      newReq = false;
    }
  } else if (window.ActiveXObject) {
    try {
      newReq = new ActiveXObject("Msxml2.XMLHTTP");
    } catch (e) {
      try {
        newReq = new ActiveXObject("Microsoft.XMLHTTP");
      } catch (e) {
        newReq = false;
      }
    }
  }
  return newReq;
}

function getCursorPos(campo) {
  if (document.selection) {
    campo.focus();
    var oSel = document.selection.createRange();
    oSel.moveStart('character', -campo.value.length);
    campo.selectionEnd = oSel.text.length;
    oSel.setEndPoint('EndToStart', document.selection.createRange());
    campo.selectionStart = oSel.text.length;
  }
  return {start: campo.selectionStart, end: campo.selectionEnd};
}

function putCursorInPos(campo, pos, npos)   // npos = 0..nº de caracteres
{
  if (typeof document.selection != 'undefined' && document.selection) { // IE
    var tex = campo.value;
    campo.value = '';
    campo.focus();
    var str = document.selection.createRange();
    campo.value = tex;
    str.move("character", pos);
    str.moveEnd("character", npos);
    str.select();
  } else if (typeof campo.selectionStart != 'undefined') {
    campo.setSelectionRange(pos, pos + npos);
    focus();
  }
}

function fmtDecimal(obj, decimais) {
  if (obj.value != "") {
    var sValor = obj.value;
    sValor = sValor.replace(".", "").replace(",", ".");
    var valor = parseFloat(sValor);
    var partes = valor.toFixed(decimais).split('.');
    partes[0] = partes[0].split(/(?=(?:...)*$)/).join('.');
    obj.value = partes.join(",");
  }
}

function fmtStrNum(obj, decimais) {
  if (obj.value != "")
    fmtDecimal(obj, decimais);
}

function fmtCNPJ(cnpj) {
  if (cnpj.value == "") return "";
  var numero = desformatar(cnpj.value);
  numero = numero.lpad(14, "0");
  reCnpj = /(\d{2})(\d{3})(\d{3})(\d{4})(\d{2})$/;
  numero = numero.replace(reCnpj, "$1.$2.$3/$4-$5");
  cnpj.value = numero;
}

function fmtCPF(cpf) {
  if (cpf.value == "") return "";
  var numero = desformatar(cpf.value);
  numero = numero.lpad(11, "0");
  reCpf = /(\d{3})(\d{3})(\d{3})(\d{2})$/;
  numero = numero.replace(reCpf, "$1.$2.$3/$4");
  cpf.value = numero;
}

function fmtCPFCNPJ(campo) {
  if (campo.value == "") return "";
  var numero = desformatar(campo.value);
  if (numero.length < 12)
    fmtCPF(campo);
  else if (numero.length < 15)
    fmtCNPJ(campo);
}

function isRoman(valor) {
  var tabDigitos = ["C", "D", "I", "L", "M", "V", "X"];
  var val = valor.toUpperCase();
  var letra = "";
  for (j = 0; j < val.length; j++) {
    letra = val.charAt(j);
    if (!tabDigitos.inArray(letra))
      return false;
  }
  return true;
}

function textToName(campo) {
  var tabMinusculas = ["da", "de", "do", "na", "no", "das", "dos", "em"];
  var tabMaiusculas = ["fl", "ltda", "iema", "cetesb", "fepam", "fatma", "sema", "cni", "anamma", "cnc", "fecombustíveis", "s.a.",
    "inss", "previc", "light", "ampla", "simepetro", "sindicom", "simepetro", "abema", "celus", "mb",
    "sindicombustíveis", "minaspetro", "sindcomb", "sindestado", "sindipetro", "sindópolis", "sinpeb",
    "fiesp", "recap", "regran", "resan", "sincopetro", "sindirepa", "sulpetro", "sincombustíveis", "gnv",
    "br", "ac", "al", "am", "ap", "ba", "ce", "df", "es", "go", "ma", "mg", "ms", "mt",
    "pa", "pb", "pe", "pi", "pr", "rj", "rn", "ro", "rr", "rs", "sc", "se", "sp", "to"];
  var texto = campo.value.replace(/-/g, " - ");
  texto = texto.replace(/\s+/g, " ").replace(/^\s+/, "").replace(/\s+$/, "");
  texto = texto.toLowerCase();
  //var palavras = texto.split(/[(-\/: )]+/);  // a-záàâãéèêíïóôõöúçñÁÀÂÃÉÈÊÍÏÓÒÖÚÇÑ
  var palavras = texto.split(" ");
  for (i = 0; i < palavras.length; i++) {
    if ((tabMaiusculas.inArray(palavras[i])) || (isRoman(palavras[i])))
      palavras[i] = palavras[i].toUpperCase();
    else if ((tabMinusculas.inArray(palavras[i])) || (palavras[i].length == 1))
      palavras[i] = palavras[i].toLowerCase();
    else
      palavras[i] = palavras[i][0].toUpperCase() + palavras[i].substring(1, palavras[i].length);
  }
  texto = palavras.join(" ");
  texto = texto.replace(/D'o/, "d'O").replace(/D'a/, "d'A").replace(/D'á/, "d'Á");
  campo.value = texto;
}

function soCoordenadas(campo) {
  let s = "";
  let dig;
  let jaTemPonto = false;
  let jaTemSinal = false;
  vr = campo.value;
  tam = vr.length;
  for (i = 0; i < tam; i++) {
    dig = vr.substring(i, i + 1);
    if ((dig >= "0" && dig <= "9") || (dig == "." && !jaTemPonto) || (dig == "-" && !jaTemSinal)) {
      s = s + dig;
      if (dig == ".")
        jaTemPonto = true;
      if (dig == "-")
        jaTemSinal = true;
    }
  }
  campo.value = s;
  return s;
}

function verificarEmail(campo) {
  var padrao = /^[a-zA-Z0-9._-]+@([a-zA-Z0-9.-]+\.)+[a-zA-Z0-9.-]{2,}$/;
  var emails = campo.value;
  var email = emails.split(",");
  for (i = 0; i < email.length; i++) {
    email[i] = email[i].trim();
    if ((email[i] != "") && (!email[i].match(padrao))) {
      showMensagemFocus("E-mail inválido: " + email[i] + ".", campo);
      return false;
    }
  }
  return true;
}

/**
 * Funções de formatação de campos
 *
 * mascara(formato, ekeypress, objeto, browser)
 * uso: onkeypress="mascara('data', event, this, 'IE')"
 *
 * formatar(campo, mascara)
 * uso: onkeypress="formatar(this, '##/##/####')"
 *
 * function fmtDado(campo, evento, mascara)
 * uso: onkeypress="fmtDado(this, event, '(##)####-####')"
 *
 * fmtValor(campo, evento)
 * uso: onkeypress="return fmtValor(this,event)"
 *
 */
function mascara(formato, ekeypress, objeto, browser) {
  campo = eval(objeto);

  if (browser == "Firefox")
    keypress = ekeypress.charCode
  else
    keypress = ekeypress.keyCode;

  // mascara para data dd/mm/aaaa
  if (formato == 'data') {
    caracteres = '1234567890';
    separacoes = 2;
    separacao1 = '/';
    separacao2 = '/';
    conjunto1 = 2;
    conjunto2 = 2;
    conjunto3 = 4;

    if (((keypress >= 48) && (keypress <= 57)) && campo.value.length < (conjunto1 + conjunto2 + conjunto3 + separacoes)) {
      if (campo.value.length > 8) {
        //colocar aqui o campo seguinte que será focado após digitar a data, ou comentá-lo
        //document.frm.nome.focus();
      }
      if (campo.value.length == conjunto1)
        campo.value = campo.value + separacao1;
      if (campo.value.length == conjunto1 + conjunto2 + 1)
        campo.value = campo.value + separacao2;
    } else event.returnValue = false;
  }

  // mascara para hora hh:mm
  if (formato == 'hora') {
    caracteres = '1234567890';
    separacoes = 1;
    separacao1 = ':';
    conjunto1 = 2;
    conjunto2 = 2;

    if (((keypress >= 48) && (keypress <= 57)) && campo.value.length < (conjunto1 + conjunto2 + separacoes)) {
      if (campo.value.length > 4) {
        //colocar aqui o campo seguinte que será focado após digitar a hora, ou comentá-lo
        //document.frm.nome.focus();
      }
      if (campo.value.length == conjunto1)
        campo.value = campo.value + separacao1;
    } else
      event.returnValue = false;
  }
}

function formatar(src, mask) {
  var i = src.value.length;
  var saida = '#';
  var texto = mask.substring(i)
  if (texto.substring(0, 1) != saida)
    src.value += texto.substring(0, 1);
}

  if (isMS) {
    keypress = evento.keyCode;
  } else {
    keypress = evento.charCode;
  }

  if (((keypress >= 48) && (keypress <= 57)) || (!isMS)) {
    var i = src.value.length;

    var texto = mask.substring(i);
    if (texto.substring(0, 1) != saida) {
      src.value += texto.substring(0, 1);
    }
  } else
    evento.keyCode = 0;
}

function fmtValor(fld, e) {
  var milSep = '.';
  var decSep = ',';
  var sep = 0;
  var key = '';
  var i = j = 0;
  var len = len2 = 0;
  var strCheck = '0123456789';
  var aux = aux2 = '';

  // if (navigator.appName.substring(0,9) != "Microsoft") return true;

  var whichCode = (window.Event) ? e.which : e.keyCode;
  if ((whichCode == 13) || (whichCode == 8) || (whichCode == 0)) return true; // Entra

  key = String.fromCharCode(whichCode); // Comece o valor chave do código chave
  if (strCheck.indexOf(key) == -1) return false; // Chave inválida

  len = fld.value.length;
  for (i = 0; i < len; i++)
    if ((fld.value.charAt(i) != '0') && (fld.value.charAt(i) != decSep)) break;
  aux = '';
  for (; i < len; i++)
    if (strCheck.indexOf(fld.value.charAt(i)) != -1) aux += fld.value.charAt(i);
  aux += key;
  len = aux.length;
  if (len == 0) fld.value = '';
  if (len == 1) fld.value = '0' + decSep + '0' + aux;
  if (len == 2) fld.value = '0' + decSep + aux;
  if (len > 2) {
    aux2 = '';
    for (j = 0, i = len - 3; i >= 0; i--) {
      if (j == 3) {
        aux2 += milSep;
        j = 0;
      }
      aux2 += aux.charAt(i);
      j++;
    }
    fld.value = '';
    len2 = aux2.length;
    for (i = len2 - 1; i >= 0; i--)
      fld.value += aux2.charAt(i);
    fld.value += decSep + aux.substring(len - 2, len);
  }
  return false;
}

function floatToNum(numero, decimais) {
  var numero = numero.toFixed(decimais).split('.');
  numero[0] = numero[0].split(/(?=(?:...)*$)/).join('.');
  return numero.join(',');
}

function dolarToReal(numero) {
  var numero = numero.toFixed(2).split('.');
  numero[0] = numero[0].split(/(?=(?:...)*$)/).join('.');
  return numero.join(',');
}

function realToDolar(numero) {
  var numero = numero.replace(".", "");
  numero = numero.replace(",", ".");
  return numero;
}

/**
 * Multiplica dígitos vezes posições
 *
 * @param string digitos Os digitos desejados
 * @param string posicoes A posição que vai iniciar a regressão
 * @param string soma_digitos A soma das multiplicações entre posições e dígitos
 * @return string Os dígitos enviados concatenados com o último dígito
 */

function calcDigitosPosicoes(digitos, posicoes, soma_digitos) {
  // Garante que o valor é uma string
  digitos = digitos.toString();

  // Faz a soma dos dígitos com a posição
  // Ex. para 10 posições:
  //   0    2    5    4    6    2    8    8   4
  // x10   x9   x8   x7   x6   x5   x4   x3  x2
  //   0 + 18 + 40 + 28 + 36 + 10 + 32 + 24 + 8 = 196
  for (var i = 0; i < digitos.length; i++) {
    soma_digitos = soma_digitos + (digitos[i] * posicoes); // Preenche a soma com o dígito vezes a posição
    posicoes--;  // Subtrai 1 da posição
    // Parte específica para CNPJ  Ex.: 5-4-3-2-9-8-7-6-5-4-3-2
    if (posicoes < 2) {  // Retorno a posição para 9
      posicoes = 9;
    }
  }
  // Captura o resto da divisão entre soma_digitos dividido por 11 Ex.: 196 % 11 = 9
  soma_digitos = soma_digitos % 11;
  if (soma_digitos < 2) { // Verifica se soma_digitos é menor que 2
    soma_digitos = 0; // soma_digitos agora será zero
  } else {
    // Se for maior que 2, o resultado é 11 menos soma_digitos Ex.: 11 - 9 = 2
    // Nosso dígito procurado é 2
    soma_digitos = 11 - soma_digitos;
  }
  // Concatena mais um dígito aos primeiro nove dígitos Ex.: 025462884 + 2 = 0254628842
  var cpf = digitos + soma_digitos;
  return cpf;   // Retorna
} // calcDigitosPosicoes

/**
 * Valida se for CPF
 *
 * @cpf  string cpf O CPF com ou sem pontos e traço
 * @return bool True para CPF correto - False para CPF incorreto
 */
function isCPF(cpf) {
// 1. Remove qualquer caractere que não seja número
  const cpfLimpo = cpf.toString().replace(/\D/g, '');

  // 2. O CPF deve ter exatamente 11 dígitos
  if (cpfLimpo.length !== 11) return false;

  // 3. Elimina CPFs com todos os números iguais (ex: 111.111.111-11)
  if (/^(\d)\1+$/.test(cpfLimpo)) return false;

  // 4. Validação do primeiro dígito verificador
  let soma = 0;
  let resto;

  for (let i = 1; i <= 9; i++) {
    soma += parseInt(cpfLimpo.substring(i - 1, i)) * (11 - i);
  }

  resto = (soma * 10) % 11;
  if ((resto === 10) || (resto === 11)) resto = 0;
  if (resto !== parseInt(cpfLimpo.substring(9, 10))) return false;

  // 5. Validação do segundo dígito verificador
  soma = 0;
  for (let i = 1; i <= 10; i++) {
    soma += parseInt(cpfLimpo.substring(i - 1, i)) * (12 - i);
  }

  resto = (soma * 10) % 11;
  if ((resto === 10) || (resto === 11)) resto = 0;
  if (resto !== parseInt(cpfLimpo.substring(10, 11))) return false;

  return true;
}

/**
 * Valida se for um CNPJ
 *
 * @cnpj string cnpj
 * @return bool true para CNPJ correto
 */
function isCNPJ(cnpj) {
  // 1. Remove pontuação e espaços, deixando apenas letras e números
  const cnpjLimpo = cnpj.toString().replace(/[^A-Za-z0-9]/g, '').toUpperCase();

  // 2. O CNPJ deve ter exatamente 14 caracteres
  if (cnpjLimpo.length !== 14) return false;

  // 3. Elimina sequências idênticas conhecidas
  if (/^([A-Z0-9])\1+$/.test(cnpjLimpo)) return false;

  // Separar o corpo dos dois dígitos verificadores numéricos
  const corpo = cnpjLimpo.substring(0, 12);
  const digitosMod0 = cnpjLimpo.substring(12, 14);

  // Garantir que os dois últimos caracteres sejam números
  if (!/^\d{2}$/.test(digitosMod0)) return false;

  // 4. Função auxiliar para calcular cada dígito verificador
  function calcularDigito(base, pesos) {
    let soma = 0;

    for (let i = 0; i < base.length; i++) {
      // Converte o caractere usando a tabela ASCII (Valor ASCII - 48)
      const valorCaractere = base.charCodeAt(i) - 48;
      soma += valorCaractere * pesos[i];
    }

    const resto = soma % 11;
    return resto < 2 ? 0 : 11 - resto;
  }

  // Pesos oficiais para o cálculo do CNPJ
  const pesosDigito1 = [5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2];
  const pesosDigito2 = [6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2];

  // 5. Validar o primeiro dígito
  const digito1 = calcularDigito(corpo, pesosDigito1);
  if (digito1 !== parseInt(cnpjLimpo.charAt(12))) return false;

  // 6. Validar o segundo dígito (inclui o primeiro dígito calculado no corpo)
  const digito2 = calcularDigito(corpo + digito1, pesosDigito2);
  if (digito2 !== parseInt(cnpjLimpo.charAt(13))) return false;

  return true;
}

function campoZerado(campo, posicoes) {
  let valor = campo.value.trim();
  valor = valor.replace(/[^0-9]/g, '');

  if (valor == "")
    return false;
  let tam = valor.length;
  valor = "0".repeat(posicoes - tam) + valor;
  if (valor == "0".repeat(posicoes))
    return true;
  else
    return false;
}

function getTextBetween(str, tokenIni, tokenFim) {
  const startIndex = str.indexOf(tokenIni) + tokenIni.length;
  const endIndex = str.indexOf(tokenFim);
  return str.substring(startIndex, endIndex);
}
