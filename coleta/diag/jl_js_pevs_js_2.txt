function tratarPevs(estado)
{
 $.post("pevs_tratar.php", 
         { estado: estado }, 
	 function(dados)
	 {
		$("#estado-selecionado").html(dados);
	 });
}

    $(function() {
        // selecionar estado
        $('.a-estado').click(function() {
            $('#btn-estado').text($(this).text())
            let uf = $(this).attr('data-uf');
            tratarPevs(uf);
            $('#estado').removeClass('show')
            $('#estado-selecionado').addClass('show')
            $('#nome-estado').text($(this).text())
        })
        // btn estado
        $('#btn-estado').click(function() {
            $('#btn-estado').text('SELECIONE O ESTADO')
            $('#estado-selecionado').removeClass('show')
        })
    })
